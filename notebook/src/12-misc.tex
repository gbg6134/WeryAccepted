\section{Searching}

\entry{Binary search on the answer}{$O(\log)$}{%
No template needed. The predicate must be monotone false$\dots$false
true$\dots$true; keep the invariant \code{pred(lo)} false, \code{pred(hi)} true and
loop \code{while (hi - lo > 1)} with \code{m = lo + (hi-lo)/2}; the answer is
\code{hi}. On a sorted array use \code{lower\_bound}/\code{upper\_bound}, or
\code{partition\_point(all(v), pred)}.\\[0.3ex]
Almost every ``minimise the maximum'' / ``can we do it within $k$'' problem is this
plus a greedy check. Reals: fix the iteration count (100 halvings) instead of looping
on a tolerance, which can spin forever.}

\entry{Ternary search}{$O(\log(b-a))$}{%
From KACTL. \code{ternSearch(a, b, f)}: the smallest \code{i} in \code{[a, b]} that
\textbf{maximises} \code{f(i)}, assuming $f(a) < \dots < f(i) \ge \dots \ge f(b)$. To
\textbf{minimise}, change the \code{<} at (A) and at (B) to \code{>}. To allow flat
stretches on the other side, make (A) \code{<=} and reverse the loop at
(B).\\[0.3ex]
Usage: \code{int ind = ternSearch(0, n-1, [\&](int i)\{ return a[i]; \});}. Reals:
the classic thirds loop, 200 iterations, on \code{ld}.}

%LABEL ternSearch
\begin{lstlisting}
template<class F>
int ternSearch(int a, int b, F f) {
    assert(a <= b);
    while (b - a >= 5) {
        int mid = (a + b) / 2;
        if (f(mid) < f(mid+1)) a = mid; // (A)
        else b = mid+1;
    }
    rep(i,a+1,b+1) if (f(a) < f(i)) a = i; // (B)
    return a;
}
\end{lstlisting}

\entry{Coordinate compression}{$O(n \log n)$}{%
Maps arbitrary values onto \code{0 .. k-1} keeping order, so a \code{BIT} or \code{Seg}
can index them. This is what lets you avoid \code{SparseSeg} whenever the whole input is
readable up front --- which is almost always.\\[0.3ex]
Remember to compress \emph{every} value you will ever look up, queries included, not
just the ones in the array. For half-open interval problems add both \code{l} and
\code{r} (and sometimes \code{r+1}).}

\begin{lstlisting}
// returns the sorted distinct values; id(v) is v's index
vi compress(vi vals) {
    sort(all(vals));
    vals.erase(unique(all(vals)), vals.end());
    return vals;
}
int id(const vi& c, ll v) {
    return (int)(lower_bound(all(c), v) - c.begin());
}
\end{lstlisting}

\entry{Interval cover}{$O(n \log n)$}{%
Fewest intervals from \code{I} that together cover the target \code{G}; returns their
indices, or empty if impossible. Intervals are \textbf{half-open} \code{[a, b)}; for
closed ones on integers use \code{[a, b+1)}. Greedy: from the current point, among the
intervals starting at or before it, take the one that reaches furthest. Same idea for
``fewest jumps'' and ``fewest refuelling stops'' problems. From KACTL.}

%LABEL cover
\begin{lstlisting}
template<class T>
vector<int> cover(pair<T,T> G, vector<pair<T,T>> I) {
    vector<int> S(I.size()), R;
    iota(all(S), 0);
    sort(all(S), [&](int a, int b) { return I[a] < I[b]; });
    T cur = G.first;
    int at = 0;
    while (cur < G.second) {
        pair<T, int> mx = make_pair(cur, -1);
        while (at < (int)I.size() && I[S[at]].first <= cur) {
            mx = max(mx, make_pair(I[S[at]].second, S[at]));
            at++;
        }
        if (mx.second == -1) return {};
        cur = mx.first;
        R.push_back(mx.second);
    }
    return R;
}
\end{lstlisting}
